        \item {\large \textbf{Singleton} — فقط یک نمونه}

\slide{
    \slidetitle{Factory Method}
    \vspace{0.5cm}
    \begin{center}
        {\large\bfseries ساخت اشیاء بدون دانستن کلاس دقیق}
    \end{center}
    \vspace{0.8cm}
    {\large\textbf{مسئله:}}
    \begin{itemize}[itemsep=0.3em]
        \item کلاینت نمی‌داند کدام کلاس را بسازد
        \item تصمیم ساخت باید در یک جا باشد
    \end{itemize}
    \vspace{0.5cm}
    {\large\textbf{راه‌حل:}}
    \begin{itemize}[itemsep=0.3em]
        \item یک interface مشترک
        \item یک Factory که شیء مناسب را می‌سازد
    \end{itemize}
}

\slide{
    \slidetitle{Factory — مثال HomeStore}
    \vspace{0.3cm}
    {\large\textbf{مسئله:} روش‌های پرداخت مختلف}
    \vspace{0.3cm}
    \begin{center}
        \begin{tabular}{lp{8cm}}
            \toprule
            \textbf{کلاس} & \textbf{نقش} \\
            \midrule
            \lr{PaymentMethod} & interface مشترک \\
            \hline
            \lr{CardPayment} & پرداخت با کارت \\
            \hline
            \lr{PayPalPayment} & پرداخت با PayPal \\
            \hline
            \lr{PaymentFactory} & ساخت روش مناسب \\
            \bottomrule
        \end{tabular}
    \end{center}
    \vspace{0.5cm}
    {\large\textbf{مزیت:} افزودن روش جدید = فقط کلاس جدید}
}

\slide{
    \slidetitle{Abstract Factory}
    \vspace{0.5cm}
    \begin{center}
        {\large\bfseries ساخت خانواده‌ای از اشیاء مرتبط}
    \end{center}
    \vspace{0.8cm}
    {\large\textbf{مثال:}}
    \begin{itemize}[itemsep=0.3em]
        \item \lr{IUIFactory} — ساخت دکمه، فرم، جدول
        \item \lr{LightThemeFactory} — نسخهٔ روشن
        \item \lr{DarkThemeFactory} — نسخهٔ تیره
    \end{itemize}
    \vspace{0.5cm}
    {\large\textbf{تفاوت با Factory Method:}}
    \begin{itemize}[itemsep=0.3em]
        \item Factory Method: یک شیء
        \item Abstract Factory: خانواده‌ای از اشیاء
    \end{itemize}
}

\slide{
    \slidetitle{Singleton}
    \vspace{0.5cm}
    \begin{center}
        {\large\bfseries فقط یک نمونه از کلاس}
    \end{center}
    \vspace{0.8cm}
    {\large\textbf{کاربرد:}}
    \begin{itemize}[itemsep=0.3em]
        \item اتصال به دیتابیس
        \item تنظیمات برنامه
        \item لاگ کردن
    \end{itemize}
    \vspace{0.5cm}
    {\large\textbf{هشدار:}}
    \begin{itemize}[itemsep=0.3em]
        \item تست سخت می‌شود
        \item Coupling بالا
        \item استفاده با احتیاط
    \end{itemize}
}

\slide{
    \slidetitle{Builder}
    \vspace{0.5cm}
    \begin{center}
        {\large\bfseries ساخت گام‌به‌گام اشیاء پیچیده}
    \end{center}
    \vspace{0.8cm}
    {\large\textbf{مثال:} ساخت سفارش}
    \vspace{0.5cm}
    \begin{itemize}[itemsep=0.3em]
        \item \lr{OrderBuilder.addItem()}
        \item \lr{OrderBuilder.setAddress()}
        \item \lr{OrderBuilder.setPayment()}
        \item \lr{OrderBuilder.build()}
    \end{itemize}
    \vspace{0.5cm}
    {\large\textbf{مزیت:} کد خواناتر از constructor با ۱۰ پارامتر}
}

\slide{
    \slidetitle{Structural Patterns — مقدمه}
    \vspace{0.5cm}
    \begin{center}
        {\large\bfseries الگوهای ترکیب اشیاء}
    \end{center}
    \vspace{0.8cm}
    \begin{itemize}[itemsep=0.5em]
        \item {\large \textbf{Adapter} — تبدیل interface}
        \item {\large \textbf{Decorator} — افزودن رفتار}
        \item {\large \textbf{Facade} — ساده‌سازی}
        \item {\large \textbf{Proxy} — کنترل دسترسی}
        \item {\large \textbf{Composite} — ساختار درختی}
    \end{itemize}
}

\slide{
    \slidetitle{Adapter}
    \vspace{0.5cm}
    \begin{center}
        {\large\bfseries تبدیل interface ناسازگار}
    \end{center}
    \vspace{0.8cm}
    {\large\textbf{مثال:} درگاه پرداخت خارجی}
    \begin{itemize}[itemsep=0.3em]
        \item کد ما: \lr{PaymentGateway.pay()}
        \item درگاه: \lr{StripeAPI.charge()}
        \item Adapter: \lr{StripeAdapter implements PaymentGateway}
    \end{itemize}
    \vspace{0.5cm}
    {\large\textbf{مزیت:} کد ما از جزئیات درگاه مستقل می‌شود}
}

\slide{
    \slidetitle{Decorator}
    \vspace{0.5cm}
    \begin{center}
        {\large\bfseries افزودن رفتار بدون تغییر کلاس}
    \end{center}
    \vspace{0.8cm}
    {\large\textbf{مثال:} لاگ کردن}
    \begin{itemize}[itemsep=0.3em]
        \item \lr{OrderService} — اصلی
        \item \lr{LoggedOrderService} — با لاگ
        \item \lr{CachedOrderService} — با کش
    \end{itemize}
    \vspace{0.5cm}
    {\large\textbf{مزیت:} ترکیب رفتارها بدون تغییر کد اصلی}
}

\slide{
    \slidetitle{Facade}
    \vspace{0.5cm}
    \begin{center}
        {\large\bfseries ساده‌سازی interface پیچیده}
    \end{center}
    \vspace{0.8cm}
    {\large\textbf{مثال:} فرآیند خرید}
    \begin{itemize}[itemsep=0.3em]
        \item \lr{OrderService}, \lr{PaymentService}, \lr{EmailService}, 
\lr{StockService}
        \item Facade: \lr{PurchaseFacade.buy()}
    \end{itemize}
    \vspace{0.5cm}
    {\large\textbf{مزیت:} کلاینت فقط یک متد صدا می‌زند}
}

\slide{
    \slidetitle{Behavioral Patterns — مقدمه}
    \vspace{0.5cm}
    \begin{center}
        {\large\bfseries الگوهای تعامل اشیاء}
    \end{center}
    \vspace{0.8cm}
    \begin{itemize}[itemsep=0.5em]
        \item {\large \textbf{Strategy} — الگوریتم قابل تعویض}
        \item {\large \textbf{Observer} — اطلاع‌رسانی}
        \item {\large \textbf{Command} — کپسوله‌سازی درخواست}
        \item {\large \textbf{State} — تغییر رفتار با حالت}
        \item {\large \textbf{Template Method} — قالب الگوریتم}
    \end{itemize}
}

\slide{
    \slidetitle{Strategy}
    \vspace{0.5cm}
    \begin{center}
        {\large\bfseries الگوریتم قابل تعویض}
    \end{center}
    \vspace{0.8cm}
    {\large\textbf{مسئله:} محاسبهٔ هزینهٔ ارسال}
    \begin{itemize}[itemsep=0.3em]

    \end{itemize}
    \vspace{0.5cm}
    {\large\textbf{راه‌حل:} interface \lr{ShippingStrategy} + سه 
implementation}
}

\slide{

    \vspace{0.5cm}

        {\large\bfseries اطلاع‌رسانی تغییرات}
    \end{center}
    \vspace{0.8cm}

\slide{
    \slidetitle{Repository Pattern}

    \end{center}
    \vspace{0.8cm}

}
