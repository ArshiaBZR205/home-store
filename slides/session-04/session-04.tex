\slide{
    \slidetitle{Use Case Diagram}
    \vspace{0.5cm}
    {\large چه کسی با سیستم چه کاری می‌کند؟}
    \vspace{1cm}
    \begin{itemize}[itemsep=0.5em]
        \item \textbf{Actor:} کاربر یا سیستم خارجی
        \item \textbf{Use Case:} یک عملکرد
        \item \textbf{Relationship:} ارتباط
    \end{itemize}
    \vspace{1cm}
    \begin{center}
        {\myaccent{مثال: مشتری → مشاهدهٔ محصولات}}
    \end{center}
}

\slide{
    \slidetitle{Class Diagram}
    \vspace{0.5cm}
    {\large ساختار کلاس‌ها و روابط}
    \vspace{1cm}
    \begin{itemize}[itemsep=0.5em]
        \item \textbf{Class:} نام، ویژگی‌ها، متدها
        \item \textbf{Association:} ارتباط ساده
        \item \textbf{Inheritance:} ارث‌بری
        \item \textbf{Composition:} ترکیب
    \end{itemize}
    \vspace{0.5cm}
    \begin{center}
        {\myaccent{مثال: Product، Order، Customer}}
    \end{center}
}

\slide{
    \slidetitle{Sequence Diagram}
    \vspace{0.5cm}
    {\large ترتیب تعاملات در زمان}
    \vspace{1cm}
    \begin{itemize}[itemsep=0.5em]
        \item \textbf{Actor:} کاربر
        \item \textbf{Object:} شیء
        \item \textbf{Message:} پیام
        \item \textbf{Lifeline:} خط زمان
    \end{itemize}
    \vspace{0.5cm}
    \begin{center}
        {\myaccent{مثال: مشتری → سبد خرید → ثبت سفارش}}
    \end{center}
}

        \item \textbf{Fork / Join:} انشعاب
    \end{itemize}
    \vspace{0.5cm}
    \begin{center}
        {\myaccent{مثال: فرآیند خرید}}
